# Query modes
QUERY_MODES = {
    "simple_rag": "Simple RAG (Vector Search Only)",
    "graph_rag": "GraphRAG (Concept-Aware)",
    "graph_deep": "GraphRAG with Prerequisites"
}

def simple_rag_answer(query):
    """Traditional RAG approach using vector search only"""
    try:
        # Get relevant chunks from vector DB
        chunks = query_vector_db(query, n_results=5)
        # Build prompt
        prompt = build_prompt(query, chunks)
        # Get answer from LLM
        answer = ask_ollama(prompt)
        return answer
    except Exception as e:
        print(f"Simple RAG error: {e}")
        return f"Error in simple RAG: {str(e)}"

def graph_rag_answer(query, include_prereqs=False):
    """GraphRAG approach using knowledge graph"""
    try:
        if not kg_instance:
            return "Knowledge Graph not available. Using simple RAG instead."
        
        # 1. Find relevant concepts from the query
        concept_matches = kg_instance.concepts_for_query(query, top_k=4)
        
        if not concept_matches:
            return "No relevant concepts found in knowledge base."
        
        # Extract concept IDs
        concept_ids = [cid for cid, score in concept_matches]
        
        # 2. Get subgraph for these concepts
        prereq_depth = 2 if include_prereqs else 0
        subgraph = kg_instance.subgraph_for_concepts(
            concept_ids, 
            prereq_depth=prereq_depth,
            include_resources=True,
            include_examples=True
        )
        
        # 3. Build specialized prompt with graph context
        # First get definitions and resources
        context_parts = []
        
        # Add concept definitions
        context_parts.append("### RELEVANT CONCEPTS ###")
        for cid in concept_ids:
            node_data = kg_instance.G.nodes[cid]
            title = node_data.get('title', cid)
            definitions = node_data.get('definitions', [])
            if definitions:
                context_parts.append(f"• {title}: {definitions[0]}")
        
        # Add resources (chunks) connected to these concepts
        context_parts.append("\n### RELEVANT RESOURCES ###")
        resource_count = 0
        for cid in concept_ids:
            for u, v, ed in kg_instance.G.in_edges(cid, data=True):
                if ed.get("type") == "explains":
                    resource_node = kg_instance.G.nodes[u]
                    if resource_node.get('type') == 'resource':
                        text_preview = resource_node.get('text', '')[:300]
                        if text_preview:
                            context_parts.append(f"• Resource excerpt: {text_preview}...")
                            resource_count += 1
                        if resource_count >= 5:  # Limit resources
                            break
            if resource_count >= 5:
                break
        
        # 4. Build final prompt
        context = "\n".join(context_parts)
        final_prompt = f"""
You are an AI tutor for an NYU AI course. Use ONLY the provided context to answer.

{context}

Question: {query}

Answer the question clearly and concisely using only the context above. 
If the answer cannot be found in the context, say "I don't have enough information to answer this question based on the course material."

Answer:
"""
        
        # 5. Get answer from LLM
        answer = ask_ollama(final_prompt)
        
        # 6. Add debugging info (could be removed in production)
        debug_info = f"\n\n[GraphRAG Info: Found {len(concept_matches)} concepts, {resource_count} resources, {len(subgraph.nodes())} nodes in subgraph]"
        
        return answer + debug_info
        
    except Exception as e:
        print(f"GraphRAG error: {e}")
        return f"Error in GraphRAG: {str(e)}"

def answer_query(query, mode="graph_rag"):
    """Main query function with mode selection"""
    if mode == "simple_rag":
        return simple_rag_answer(query)
    elif mode == "graph_rag":
        return graph_rag_answer(query, include_prereqs=False)
    elif mode == "graph_deep":
        return graph_rag_answer(query, include_prereqs=True)
    else:
        return graph_rag_answer(query, include_prereqs=False)

@app.route("/")
def index():
    return render_template("latest_graphrag_front.html", modes=QUERY_MODES)

@app.route("/ask", methods=["POST"])
def ask():
    data = request.get_json()
    user_input = data.get("query", "")
    mode = data.get("mode", "graph_rag")  # Get query mode
    
    if not user_input:
        return jsonify({"error": "No query provided"}), 400
    
    print(f"Processing query: '{user_input}' with mode: {mode}")
    
    # Get answer based on selected mode
    try:
        answer = answer_query(user_input, mode=mode)
    except Exception as e:
        print(f"Pipeline error: {e}")
        import traceback
        traceback.print_exc()
        answer = f"Sorry, I ran into an issue processing that question. Error: {str(e)}"
    
    # Save to chat history
    chat_history.append({
        "user": user_input,
        "bot": answer,
        "mode": mode
    })
    
    # Keep only last 20 messages
    if len(chat_history) > 20:
        chat_history.pop(0)
    
    return jsonify({
        "answer": answer,
        "history": chat_history,
        "mode_used": mode
    })

@app.route("/concepts", methods=["GET"])
def list_concepts():
    """API endpoint to list all concepts in the knowledge graph"""
    try:
        if not kg_instance:
            return jsonify({"error": "Knowledge Graph not available"}), 503
        
        concepts = []
        for node, data in kg_instance.G.nodes(data=True):
            if data.get('type') == 'concept':
                concepts.append({
                    "id": node,
                    "title": data.get('title', ''),
                    "definition": data.get('definitions', [''])[0][:200] + "...",
                    "connections": kg_instance.G.degree(node)
                })
        
        # Sort by number of connections
        concepts.sort(key=lambda x: x['connections'], reverse=True)
        
        return jsonify({
            "concepts": concepts[:50],  # Top 50
            "total_concepts": len(concepts)
        })
    except Exception as e:
        return jsonify({"error": str(e)}), 500

@app.route("/graph_info", methods=["GET"])
def graph_info():
    """API endpoint to get graph statistics"""
    try:
        if not kg_instance:
            return jsonify({"error": "Knowledge Graph not available"}), 503
        
        # Count node types
        node_types = {}
        edge_types = {}
        
        for node, data in kg_instance.G.nodes(data=True):
            node_type = data.get('type', 'unknown')
            node_types[node_type] = node_types.get(node_type, 0) + 1
        
        for u, v, data in kg_instance.G.edges(data=True):
            edge_type = data.get('type', 'unknown')
            edge_types[edge_type] = edge_types.get(edge_type, 0) + 1
        
        return jsonify({
            "total_nodes": len(kg_instance.G.nodes()),
            "total_edges": len(kg_instance.G.edges()),
            "node_types": node_types,
            "edge_types": edge_types,
            "is_directed": kg_instance.G.is_directed()
        })
    except Exception as e:
        return jsonify({"error": str(e)}), 500
